import { parseISO, isValid, subYears, format } from 'date-fns';

function startDateValidator(req, res, next) {
  const { startDate } = req.query;
  if (startDate && !isValid(parseISO(startDate))) {
    req.query.startDate = format(subYears(new Date(), 1), 'yyyy-MM-dd');
  }
  next();
}

/**
 * This function validates start date and end date.
 * @param startDate
 * @param endDate
 * @returns {boolean}
 */
export function isvalidDates(startDate, endDate) {
  const start = new Date(startDate);
  const end = new Date(endDate);
  const nowInUTC = new Date(formatWithTimezone({ date: Date.now() }));

  const validStartDate = differenceInMonths(nowInUTC, start) <= 12 * environment.YEARS_OF_CLAIM_HISTORY;
  const validEndDate = !isAfter(start, end);

  return validStartDate && validEndDate;
}
